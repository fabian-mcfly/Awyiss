<?php declare(strict_types=1);


use Migrations\BaseMigration;


class AlterBackgroundColorOnDummyMigration extends BaseMigration {
	/**
	 * @inheritDoc
	 */
	public function change(): void {
		$table = $this->table('dummy_migration');
		$table->changeColumn('backgroundColor', 'string', [
            'default' => null,
            'limit' => 100,
            'null' => false,
        ]);
		if ($table->hasIndexByName('BY_BACKGROUNDCOLOR')) {
			$table->removeIndexByName('BY_BACKGROUNDCOLOR');
		}

		$table->update();
	}
}
